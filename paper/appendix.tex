\section{Heston Model Simulation}
\label{app:heston}

\subsection{Euler-Maruyama Discretization}

We simulate the Heston model using the Euler-Maruyama scheme with full truncation for variance positivity:

\begin{align}
    S_{t+\Delta t} &= S_t \exp\left[\left(r - \frac{v_t^+}{2}\right)\Delta t + \sqrt{v_t^+ \Delta t} \, Z_1\right] \\
    v_{t+\Delta t} &= v_t + \kappa(\theta - v_t^+)\Delta t + \xi\sqrt{v_t^+ \Delta t}\left(\rho Z_1 + \sqrt{1-\rho^2} Z_2\right)
\end{align}

where $v_t^+ = \max(v_t, 0)$ and $(Z_1, Z_2)$ are independent standard normal random variables.

\section{Loss Function Implementations}
\label{app:losses}

\subsection{CVaR Loss}

For a batch of P\&L values $\{X_i\}_{i=1}^n$ and confidence level $\alpha$:

\begin{algorithmic}
\STATE $L_i \gets -X_i$ \COMMENT{Convert to losses}
\STATE $k \gets \lceil (1-\alpha) \cdot n \rceil$
\STATE Sort $L$ in descending order
\STATE \textbf{return} $\frac{1}{k}\sum_{i=1}^k L_i$ \COMMENT{Mean of worst $k$ losses}
\end{algorithmic}

\subsection{Entropic Loss (Numerically Stable)}

Using the log-sum-exp trick:

\begin{algorithmic}
\STATE $z_i \gets -\lambda \cdot X_i$
\STATE $z_{\max} \gets \max_i z_i$
\STATE \textbf{return} $z_{\max} + \log\left(\frac{1}{n}\sum_{i=1}^n \exp(z_i - z_{\max})\right)$
\end{algorithmic}

\subsection{Trading Penalty}

\begin{equation}
    \mathcal{L}_{trade}(\delta) = \gamma \sum_{k=0}^{N-1} |\delta_{k+1} - \delta_k|
\end{equation}

with $\delta_{-1} = \delta_N = 0$ (flat initial and final positions).

\section{Statistical Methods}
\label{app:statistics}

\subsection{Bootstrap Confidence Intervals}

For a statistic $\theta$ computed on data $X = (X_1, \ldots, X_n)$:

\begin{enumerate}
    \item For $b = 1, \ldots, B$ (with $B = 10,000$):
    \begin{enumerate}
        \item Draw bootstrap sample $X^{*b} = (X_{i_1}^*, \ldots, X_{i_n}^*)$ with replacement
        \item Compute $\theta^{*b} = \theta(X^{*b})$
    \end{enumerate}
    \item Compute percentile CI: $[\theta^*_{(\alpha/2)}, \theta^*_{(1-\alpha/2)}]$
\end{enumerate}

\subsection{Paired Bootstrap Test}

For comparing statistics $\theta_A$ and $\theta_B$ on paired data:

\begin{enumerate}
    \item Compute observed difference $d_{obs} = \theta_B(X) - \theta_A(X)$
    \item For $b = 1, \ldots, B$:
    \begin{enumerate}
        \item Draw paired bootstrap sample with same indices
        \item Compute $d^{*b} = \theta_B(X^{*b}) - \theta_A(X^{*b})$
    \end{enumerate}
    \item $p$-value: $2 \cdot \min(P(d^* \leq 0), P(d^* \geq 0))$
\end{enumerate}

\subsection{Holm-Bonferroni Correction}

For $m$ hypothesis tests with $p$-values $p_1, \ldots, p_m$:

\begin{enumerate}
    \item Order $p$-values: $p_{(1)} \leq p_{(2)} \leq \cdots \leq p_{(m)}$
    \item For $k = 1, \ldots, m$:
    \begin{enumerate}
        \item If $p_{(k)} > \alpha / (m - k + 1)$: reject no more hypotheses
        \item Else: reject hypothesis $(k)$
    \end{enumerate}
\end{enumerate}

\subsection{Cohen's d Effect Size}

For paired observations $(X_i, Y_i)$:

\begin{equation}
    d = \frac{\bar{Y} - \bar{X}}{s_D}
\end{equation}

where $s_D$ is the standard deviation of differences $D_i = Y_i - X_i$.

Interpretation:
\begin{itemize}
    \item $|d| < 0.2$: negligible
    \item $0.2 \leq |d| < 0.5$: small
    \item $0.5 \leq |d| < 0.8$: medium
    \item $|d| \geq 0.8$: large
\end{itemize}

\section{Novel Approach Algorithms}
\label{app:novel_algorithms}

\subsection{W-DRO-T: Wasserstein DRO Transformer}

\textbf{Objective Function:}
\begin{equation}
    \mathcal{L}_{\text{W-DRO-T}}(\theta) = \rho_\lambda(-\text{P\&L}(\delta^\theta)) + \varepsilon \cdot \mathbb{E}\left[\|\nabla_I \rho_\lambda(\theta; I)\|_2\right]
\end{equation}

\textbf{Training Protocol:}
\begin{enumerate}[noitemsep]
    \item \textbf{Phase 1 (CVaR Pretraining):} 50 epochs, lr = $10^{-3}$, patience = 15
    \item \textbf{Phase 2 (Entropic + DRO):} 30 epochs, lr = $10^{-4}$, $\varepsilon$ annealed $0 \to 0.1$
\end{enumerate}

\textbf{Key Hyperparameters:}
\begin{itemize}[noitemsep]
    \item Robustness radius: $\varepsilon \in [0, 0.1]$ (annealed)
    \item Gradient penalty weight: 1.0
    \item Base architecture: Transformer ($d_{\text{model}}=64$, $n_{\text{heads}}=4$, $n_{\text{layers}}=3$)
\end{itemize}

\textbf{Complexity:} $O(N^2 \cdot d_{\text{model}})$ per forward pass + gradient computation overhead for DRO penalty.

\subsection{RVSN: Rough Volatility Signature Network}

\textbf{Objective Function:}
\begin{equation}
    \delta_k = f_\theta\left(\sum_{m=1}^{M} w_m(H_k, \sigma_k) \cdot \Pi_m\left(S^{(m)}(X_{[k-\tau:k]})\right)\right)
\end{equation}

\textbf{Training Protocol:}
\begin{enumerate}[noitemsep]
    \item \textbf{Stage 1 (CVaR):} 50 epochs on rough Bergomi paths ($H \in [0.05, 0.15]$)
    \item \textbf{Stage 2 (Entropic):} 30 epochs with Hurst-adaptive gating
\end{enumerate}

\textbf{Key Hyperparameters:}
\begin{itemize}[noitemsep]
    \item Signature depth: $m \in \{2, 3, 4\}$ (adaptive)
    \item Window size: $\tau = 10$ steps
    \item Hurst parameter range: $H \in [0.05, 0.5]$
    \item Compressed signature dimension: 64
\end{itemize}

\textbf{Complexity:} $O(N \cdot \tau \cdot m!)$ for signature computation.

\subsection{SAC-CVaR: CVaR-Constrained Soft Actor-Critic}

\textbf{Objective Function:}
\begin{equation}
    \max_\pi \mathbb{E}_\pi[R] + \alpha \mathcal{H}(\pi) - \lambda \cdot \max\left(0, \text{CVaR}_{0.95}(-R) - c\right)
\end{equation}

\textbf{Training Protocol:}
\begin{enumerate}[noitemsep]
    \item Initialize replay buffer, policy $\pi_\phi$, Q-networks $Q_{\theta_1}, Q_{\theta_2}$, quantile network $Z_\psi$
    \item For each episode:
    \begin{enumerate}[noitemsep]
        \item Collect transitions using $\pi_\phi$
        \item Update critics via TD learning
        \item Update quantile network for CVaR estimation
        \item Update policy via SAC objective with CVaR penalty
        \item Update Lagrange multiplier: $\lambda \gets \max(0, \lambda + \eta_\lambda(\text{CVaR} - c))$
    \end{enumerate}
\end{enumerate}

\textbf{Key Hyperparameters:}
\begin{itemize}[noitemsep]
    \item Replay buffer size: 100,000
    \item Batch size: 256
    \item Discount factor: $\gamma = 0.99$
    \item Number of quantiles: 51
    \item CVaR threshold: $c = 3.5$
    \item Lagrange learning rate: $\eta_\lambda = 10^{-4}$
\end{itemize}

\textbf{Complexity:} $O(B \cdot N \cdot d)$ per update, where $B$ is batch size.

\subsection{3SCH: Three-Stage Curriculum Hedger}

\textbf{Objective Function:}

\textit{Stage 1:} $\mathcal{L}_1 = \text{CVaR}_{0.95}(-\text{P\&L})$

\textit{Stage 2:} $\mathcal{L}_2 = \alpha(t) \cdot \text{CVaR}_{0.95} + (1-\alpha(t)) \cdot \rho_\lambda$, with $\alpha: 0.8 \to 0.2$

\textit{Stage 3:} $\mathcal{L}_3 = \rho_\lambda(-\text{P\&L}) + \gamma \sum_k |\Delta\delta_k| + \nu \cdot \text{NoBandPenalty}$

\textbf{Training Protocol:}
\begin{enumerate}[noitemsep]
    \item \textbf{Stage 1:} 50 epochs, lr = $10^{-3}$, patience = 15
    \item \textbf{Stage 2:} 20 epochs, lr = $5 \times 10^{-4}$, $\alpha$ linear anneal
    \item \textbf{Stage 3:} 30 epochs, lr = $10^{-4}$, patience = 10
\end{enumerate}

\textbf{Key Hyperparameters:}
\begin{itemize}[noitemsep]
    \item Trading penalty: $\gamma = 10^{-3}$
    \item No-trade band width: $\epsilon = 0.15$
    \item Band penalty weight: $\nu = 10^8$
    \item Mixing coefficient range: $\alpha \in [0.2, 0.8]$
\end{itemize}

\textbf{Complexity:} Same as base LSTM, with additional loss computation overhead.

\subsection{RSE: Regime-Switching Ensemble}

\textbf{Objective Function:}
\begin{equation}
    \delta_k^{\text{RSE}} = \sum_{m=1}^{M} \left(\sum_{j=1}^{K} p(r_k = j) \cdot A_{jm}\right) \cdot \delta_k^{(m)}
\end{equation}

where $A \in \mathbb{R}^{K \times M}$ is the learnable regime-model affinity matrix.

\textbf{Training Protocol:}
\begin{enumerate}[noitemsep]
    \item \textbf{Base Model Pretraining:} Train LSTM, Transformer, Signature hedgers (30 epochs each)
    \item \textbf{Gating Training (CVaR):} Freeze base models, train regime classifier + affinity (20 epochs)
    \item \textbf{Gating Fine-tuning (Entropic):} Continue gating training (30 epochs)
\end{enumerate}

\textbf{Key Hyperparameters:}
\begin{itemize}[noitemsep]
    \item Number of regimes: $K = 4$
    \item Number of base models: $M = 3$ (LSTM, Transformer, Signature)
    \item Regime features: realized vol (windows 5, 10, 20), trend, momentum, jump indicator
    \item Gating network: MLP with 64 hidden units
\end{itemize}

\textbf{Complexity:} $O(M \cdot C_{\text{base}})$ where $C_{\text{base}}$ is base model complexity.
