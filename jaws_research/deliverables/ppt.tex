\section{The Problem}

\begin{frame}{What this work asks}
\begin{itemize}
\item Deep hedging \citep{buehler2019deep} is a powerful framework, but
  almost every published evaluation:
  \begin{itemize}
    \item assumes a single calibrated Heston regime,
    \item hedges a vanilla European call,
    \item uses a single equity underlier.
  \end{itemize}
\item Real desks face exotic payoffs, multi-asset baskets, and crisis
  regimes (GFC, COVID, banking-2023).
\item \textbf{Question}: which architecture should be deployed when
  payoff, asset count and regime vary jointly?
\end{itemize}
\end{frame}

\begin{frame}{Models on the bench}
We benchmark five canonical hedgers under one shared training protocol:
\vspace{1mm}
\begin{itemize}
\item \textbf{LSTM} -- Kozyra-style two-layer recurrent baseline
\item \textbf{Transformer} -- causal-mask Transformer encoder, Buehler-style
\item \textbf{W-DRO-T} -- Transformer + Wasserstein gradient-norm penalty
\item \textbf{3SCH} -- LSTM + three-stage curriculum (CVaR/$\to$mix$\to$entropic)
\item \textbf{RSE} -- regime-switching ensemble over LSTM/Transformer/Signature
\end{itemize}
\end{frame}

\section{Mathematical setup}

\begin{frame}{Risk-sensitive hedging objective}
Discrete-time self-financing P\&L:
\[
\mathrm{PnL}(\boldsymbol\delta) = -Z + \sum_k \langle \delta_k, \Delta S_k\rangle
- c_{\mathrm{tc}}\sum_k \|\Delta\delta_k\|_1 \mathbf 1^\top S_k.
\]
We minimise a convex risk measure $\rho(-\mathrm{PnL})$:
\begin{itemize}
\item $\cvar_\alpha(L) = \E[L\mid L\ge\mathrm{VaR}_\alpha(L)]$
\item $\rho_\lambda(X) = \tfrac{1}{\lambda}\log\E[\exp(-\lambda X)]$
\end{itemize}
Hedger output is bounded: $\delta_k = \delta_{\max}\tanh(F_\theta(I_{0:k}))$.
\end{frame}

\begin{frame}{Wasserstein-DRO duality}
\begin{block}{Blanchet--Murthy (2019)}
For Lipschitz $\ell$ and $p=1$:
\[
\sup_{W_1(\QQ,\PP)\le\varepsilon}\E_\QQ[\ell] = \E_\PP[\ell] +
\varepsilon\E_\PP[\|\nabla_X\ell\|_2] + O(\varepsilon^2).
\]
\end{block}

W-DRO-T trains the entropic stage with this gradient-norm penalty;
$\varepsilon$ is annealed $0\to0.1$.  This yields ``free'' robustness to
regime-shifted distributions.
\end{frame}

\section{Tasks and Regimes}

\begin{frame}{Seven payoffs}
\begin{itemize}
\item European call, European put
\item Up-and-out barrier ($B=1.30K$)
\item Down-and-in barrier ($B=0.80K$)
\item Cash-or-nothing digital
\item Asian call (arithmetic average)
\item Basket call ($d=3$, equal weights)
\end{itemize}
\end{frame}

\begin{frame}{Six regimes + real markets}
Synthetic Heston/Bates regimes:
\begin{itemize}
\item normal US, post-COVID US, COVID US, GFC 2008 US
\item normal India, COVID India
\end{itemize}
Real-market windows on SPY/QQQ/NIFTY:
\begin{itemize}
\item GFC 2008, vol-Q4 2018, COVID acute / extended,
  inflation 2022, banking 2023
\end{itemize}
Walk-forward train/val/test on rebased $30$-day price windows.
\end{frame}

\section{Statistical Methodology}

\begin{frame}{What ``it beats the baseline'' means here}
\begin{itemize}
\item 5 seeds: $\{42, 142, 242, 342, 442\}$
\item Bootstrap CI: $2000$ resamples
\item Paired $t$-test vs.\ LSTM, per (scenario, task)
\item Holm--Bonferroni correction at $\alpha=0.05$
\item Effect sizes: Cohen's $d$ (large $|d|$ on Asian / barrier)
\end{itemize}
\end{frame}

\section{Economic, Regulatory, Managerial Implications}

\begin{frame}{Capital, hedge accounting, deployment}
\begin{itemize}
\item \textbf{FRTB IMA capital proxy}: W-DRO-T saves $20$--$25\%$ vs.\ LSTM
  in normal US, $22$--$30\%$ in COVID US.
\item \textbf{Hedge accounting (IFRS-9 / IAS-39 / Ind-AS-109)}: all five
  pass dollar-offset; W-DRO-T highest $R^2$ in COVID.
\item \textbf{Deployment}:
  \begin{itemize}
    \item US (low friction): Transformer / W-DRO-T
    \item India (high friction, NIFTY 18 bps): 3SCH or LSTM
    \item Multi-strategy desks needing seed-stability \& explainability: RSE
  \end{itemize}
\end{itemize}
\end{frame}

\section{Limitations and Future Work}

\begin{frame}{Limitations}
\begin{itemize}
\item Daily resolution masks intraday microstructure
\item Flat transaction-cost proxies (no order-book impact)
\item Synthetic-fallback path generator $\ne$ full empirical correlation
\item Training budgets capped to fit available hardware
\end{itemize}
\end{frame}

\begin{frame}{Future work}
\begin{itemize}
\item Order-book impact (Pickard \& Lawryshyn, 2025)
\item Rough-volatility (\citealp{horvath2021deep})
\item Adaptive $\varepsilon$ for W-DRO-T tied to a stress detector
\item Hybrid 3SCH-DRO and dynamic-unfreezing RSE
\end{itemize}
\end{frame}

\section{Conclusion}

\begin{frame}{Take-aways}
\begin{itemize}
\item Causal attention is the most useful inductive bias we observed.
\item Wasserstein-DRO is essentially free at inference and pays off in stress.
\item Curricula matter for stability; ensembles for variance.
\item Code, pickled metrics, figures and tables released for replication.
\end{itemize}
\end{frame}
